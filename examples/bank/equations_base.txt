' Bank stress model: base-case equations
PD = 0.004 + 0.0015*UNEMP - 0.0004*@PCY(HPI)
LGD = 0.25 - 0.004*@PCY(HPI)
DLOG(LOANS) = 0.004 + 0.002*GDPG
ECL = PD*LGD*LOANS
PROV = ECL - ECL(-1)
NII = LOANS*(0.018 + 0.25*RATE/100)
OPEX = 0.009*LOANS
NETINC = NII - PROV - OPEX
CAP = CAP(-1) + 0.7*NETINC
RWA = 0.55*LOANS
CET1 = 100*CAP/RWA
